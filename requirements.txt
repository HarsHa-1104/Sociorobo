# Python dependencies for the Voice Manager.
#
# Corrected from the old project's requirements.txt, which listed
# faster-whisper and kokoro-onnx -- Phase 1 found neither package is
# actually imported anywhere in the real pipeline (whisper.cpp and Piper
# are invoked as external subprocess binaries, installed separately by
# scripts/setup_uno_q.sh, not via pip). pygame is dropped entirely -- the
# GUI has been removed.

numpy>=2.1,<3
                        # UNO Q's package index has good numpy>=2 aarch64 wheels

requests==2.32.3
onnxruntime>=1.28,<2
PyYAML==6.0.2

# UNO Q / Python 3.13 ARM64:
# openWakeWord 0.6.0 declares a hard dependency on tflite-runtime on Linux,
# which has no aarch64/Python 3.13 wheel and is not used anyway -- this
# project only exercises openWakeWord's ONNX inference path
# (inference_framework="onnx" in config/voice_config.yaml). Confirmed on
# this exact board: `pip install openwakeword==0.6.0` (no --no-deps) fails
# to resolve for this reason. Fix: install openwakeword with --no-deps
# (scripts/setup_uno_q.sh does this), after explicitly pinning the rest of
# openwakeword's *actual* runtime deps here so they're still installed --
# `pip show openwakeword` on this board confirms it imports scipy,
# scikit-learn, and tqdm directly, none of which were previously pinned
# anywhere, so a from-scratch install of `openwakeword==0.6.0 --no-deps`
# followed by this file alone would have left those missing.
scipy==1.18.0
scikit-learn==1.9.0
tqdm==4.70.0
openwakeword==0.6.0
ai-edge-litert>=2.0.2,<3

# motion/ (HC-05 manual Bluetooth control) -- real serial I/O for
# HC05Controller (motion/hc05_controller.py). Not used by the voice
# pipeline itself; only scripts/run_motion_controller.py imports it.
pyserial==3.5
